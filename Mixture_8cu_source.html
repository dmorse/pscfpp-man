53 {
54 const double ds = RpMixtureT::ds();
55 const int np = MixtureBase<cudaReal>::nPolymer();
56 int i, j;
57 for (i = 0; i < np; ++i) {
58 for (j = 0; j < polymer(i).nBlock(); ++j) {
59 polymer(i).block(j).allocate(ds, useBatchedFFT_);
60 }
61 }
